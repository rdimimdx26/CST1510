with open("messy_lines.txt") as f:
    for line in f:
        line = line.strip()
        if line == "" or line.startswith("#"):
            continue                      

        parts = line.split(",")           

        name = parts[0].strip().upper()                 
        used = int(parts[1].split("=")[1].strip())     
        total = int(parts[2].split("=")[1].strip())     

        print(f"{name}: {used}/{total}")
